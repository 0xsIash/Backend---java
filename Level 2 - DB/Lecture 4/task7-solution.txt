SELECT name, score,
	CASE
		WHEN score >= 90 THEN 'A'
		WHEN score >=80 AND score <= 89 THEN 'B'
		WHEN score >= 70 AND score <=79 THEN 'C'
		ELSE 'F'
	END grades
FROM students;	

SELECT score,
    CASE
        WHEN score >= 60 THEN 'pass'
        ELSE 'fail'
    END AS grade
FROM STUDENTS;

    CASE TO_CHAR(SYSDATE, 'D')
        WHEN '1' THEN 'Today is Sunday'
        WHEN '2' THEN 'Today is Monday'
        WHEN '3' THEN 'Today is Tuesday'
        WHEN '4' THEN 'Today is Wednesday'
        WHEN '5' THEN 'Today is Thursday'
        WHEN '6' THEN 'Today is Friday'
        WHEN '7' THEN 'Today is Saturday'
    END AS today
FROM DUAL;
